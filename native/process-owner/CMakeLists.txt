cmake_minimum_required(VERSION 3.22)
project(esteria_process LANGUAGES CXX)
add_library(esteria_process SHARED process_owner.cpp process_usage.cpp jni_bridge.cpp)
target_compile_features(esteria_process PRIVATE cxx_std_17)
target_compile_options(esteria_process PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(esteria_process PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,noexecstack)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(esteria_process PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(esteria_process PRIVATE pthread)
    add_executable(process_owner_test process_owner.cpp process_usage.cpp process_owner_test.cpp)
    target_compile_features(process_owner_test PRIVATE cxx_std_17)
    target_compile_options(process_owner_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(process_owner_test PRIVATE pthread)
    enable_testing()
    add_test(NAME native_owner COMMAND process_owner_test)
    add_executable(process_usage_test process_owner.cpp process_usage.cpp process_usage_test.cpp)
    target_compile_features(process_usage_test PRIVATE cxx_std_17)
    target_compile_definitions(process_usage_test PRIVATE ESTERIA_USAGE_TESTING)
    target_compile_options(process_usage_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(process_usage_test PRIVATE pthread)
    add_test(NAME native_usage COMMAND process_usage_test)
    set_tests_properties(native_usage PROPERTIES TIMEOUT 30)
    set_tests_properties(native_owner PROPERTIES TIMEOUT 60)
endif()
